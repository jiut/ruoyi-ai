# A01-认证模块详细测试步骤

## A01-01_用户登录

### 📋 测试目标
验证用户登录功能的正确性，确保系统能够正确处理各种登录场景

### 🔧 测试用例设计

#### 用例1：正常登录流程
**请求信息：**
- **接口地址：** `POST {{base_url}}/auth/login`
- **请求方式：** POST
- **Content-Type：** application/json

**请求体：**
```json
{
    "username": "{{test_username}}",
    "password": "{{test_password}}",
    "captcha": "1234",
    "uuid": "test-uuid-{{$randomUUID}}"
}
```

**前置脚本：**
```javascript
// 清除可能存在的旧token
pm.environment.unset("access_token");
pm.environment.unset("refresh_token");
pm.environment.unset("current_user_id");

console.log("🔄 开始执行登录测试");
console.log("👤 用户名: " + pm.environment.get("test_username"));
```

**测试脚本：**
```javascript
pm.test("登录响应状态码检查", function () {
    pm.expect(pm.response.code).to.equal(200);
});

pm.test("登录响应格式检查", function () {
    const response = pm.response.json();
    pm.expect(response).to.have.property('code');
    pm.expect(response).to.have.property('msg');
    pm.expect(response).to.have.property('data');
});

pm.test("登录成功验证", function () {
    const response = pm.response.json();
    
    if (response.code === 200) {
        pm.expect(response.data).to.have.property('access_token');
        pm.expect(response.data).to.have.property('userInfo');
        pm.expect(response.data.userInfo).to.have.property('userId');
        pm.expect(response.data.userInfo).to.have.property('userName');
        
        // 保存token和用户信息
        pm.environment.set("access_token", response.data.access_token);
        pm.environment.set("current_user_id", response.data.userInfo.userId);
        
        if (response.data.refresh_token) {
            pm.environment.set("refresh_token", response.data.refresh_token);
        }
        
        console.log("✅ 登录成功");
        console.log("🎫 Token: " + response.data.access_token.substring(0, 20) + "...");
        console.log("👤 用户ID: " + response.data.userInfo.userId);
    } else {
        console.log("❌ 登录失败: " + response.msg);
    }
});

pm.test("Token格式验证", function () {
    const response = pm.response.json();
    if (response.code === 200 && response.data.access_token) {
        // 验证JWT token格式 (xxx.xxx.xxx)
        const token = response.data.access_token;
        const tokenParts = token.split('.');
        pm.expect(tokenParts.length).to.equal(3);
        console.log("✅ Token格式正确");
    }
});
```

**预期结果：**
- 响应状态码：200
- 响应包含有效的access_token
- 响应包含用户信息
- Token保存到环境变量

## A01-02_Token刷新测试

### 📋 测试目标
验证Token刷新机制的正确性

### 🔧 测试用例设计

#### 用例1：使用refresh_token刷新
**前置条件：**
- 已完成登录，拥有有效的refresh_token

**请求信息：**
- **接口地址：** `POST {{base_url}}/auth/refresh`
- **请求方式：** POST
- **Authorization：** Bearer {{refresh_token}}

**前置脚本：**
```javascript
const refreshToken = pm.environment.get("refresh_token");
if (!refreshToken) {
    console.log("❌ 缺少refresh_token，请先执行登录");
    throw new Error("缺少refresh_token");
}
console.log("🔄 使用refresh_token刷新token");
```

**测试脚本：**
```javascript
pm.test("Token刷新成功", function () {
    const response = pm.response.json();
    
    if (response.code === 200) {
        pm.expect(response.data).to.have.property('access_token');
        
        // 更新环境变量中的token
        const oldToken = pm.environment.get("access_token");
        const newToken = response.data.access_token;
        
        pm.environment.set("access_token", newToken);
        
        // 验证新token与旧token不同
        pm.expect(newToken).to.not.equal(oldToken);
        
        console.log("✅ Token刷新成功");
        console.log("🎫 新Token: " + newToken.substring(0, 20) + "...");
        
        if (response.data.refresh_token) {
            pm.environment.set("refresh_token", response.data.refresh_token);
        }
    } else {
        console.log("❌ Token刷新失败: " + response.msg);
    }
});

pm.test("新Token有效性验证", function () {
    const response = pm.response.json();
    if (response.code === 200) {
        const newToken = response.data.access_token;
        
        // 使用新token测试接口访问
        pm.sendRequest({
            url: pm.environment.get("base_url") + "/designer/user/bindings",
            method: 'GET',
            header: {
                'Authorization': 'Bearer ' + newToken,
                'Content-Type': 'application/json'
            }
        }, function (err, res) {
            if (!err) {
                console.log("✅ 新Token验证成功，状态码: " + res.code);
            } else {
                console.log("❌ 新Token验证失败");
            }
        });
    }
});
```

## A01-03_登出测试

### 📋 测试目标
验证用户登出功能，确保token失效

### 🔧 测试用例设计

#### 用例1：正常登出流程
**前置条件：**
- 用户已登录，拥有有效的access_token

**请求信息：**
- **接口地址：** `POST {{base_url}}/auth/logout`
- **请求方式：** POST
- **Authorization：** Bearer {{access_token}}

**前置脚本：**
```javascript
const token = pm.environment.get("access_token");
if (!token) {
    console.log("❌ 缺少access_token，请先执行登录");
    throw new Error("缺少access_token");
}
console.log("🔄 执行登出操作");
```

**测试脚本：**
```javascript
pm.test("登出响应检查", function () {
    const response = pm.response.json();
    pm.expect(response.code).to.equal(200);
    console.log("✅ 登出操作执行成功");
});

pm.test("Token失效验证", function () {
    const response = pm.response.json();
    if (response.code === 200) {
        const token = pm.environment.get("access_token");
        
        // 使用原token测试接口访问，应该失败
        pm.sendRequest({
            url: pm.environment.get("base_url") + "/designer/user/bindings",
            method: 'GET',
            header: {
                'Authorization': 'Bearer ' + token,
                'Content-Type': 'application/json'
            }
        }, function (err, res) {
            if (!err) {
                const testResponse = res.json();
                if (testResponse.code === 401 || testResponse.code === 403) {
                    console.log("✅ Token已失效，登出成功");
                } else {
                    console.log("⚠️ Token可能仍然有效，登出可能未完全生效");
                }
            }
        });
        
        // 清除环境变量
        pm.environment.unset("access_token");
        pm.environment.unset("refresh_token");
        pm.environment.unset("current_user_id");
        
        console.log("🧹 已清除token相关环境变量");
    }
});
```

#### 用例2：重复登出测试
**说明：** 连续两次调用登出接口

**测试脚本：**
```javascript
pm.test("重复登出处理", function () {
```

## 🧪 测试方法论应用

### 基于七大测试用例设计方法的分析

#### 1. 等价类划分法
**应用场景：** 用户登录的输入参数验证

**等价类划分：**
- **有效等价类：**
  - 正确的用户名格式
  - 正确的密码格式
  - 有效的验证码
- **无效等价类：**
  - 空用户名/密码
  - 错误的用户名/密码
  - 无效的验证码
  - 不存在的用户

**测试数据设计：**
```javascript
// 有效等价类测试数据
const validLoginData = [
    { username: "admin", password: "admin123", captcha: "1234" },
    { username: "test_user", password: "test123", captcha: "5678" }
];

// 无效等价类测试数据
const invalidLoginData = [
    { username: "", password: "admin123", captcha: "1234" },        // 空用户名
    { username: "admin", password: "", captcha: "1234" },           // 空密码
    { username: "nonexist", password: "admin123", captcha: "1234" }, // 不存在用户
    { username: "admin", password: "wrong", captcha: "1234" }        // 错误密码
];
```

#### 2. 边界值分析法
**应用场景：** 用户名、密码长度限制测试

**边界值测试：**

#### 自动化脚本增强：

```javascript
// 综合测试方法的验证脚本
pm.test("综合认证测试验证", function () {
    const response = pm.response.json();
    const testMethod = pm.environment.get("current_test_method");
    
    switch(testMethod) {
        case "equivalence_class":
            // 等价类测试验证逻辑
            validateEquivalenceClass(response);
            break;
        case "boundary_value": 
            // 边界值测试验证逻辑
            validateBoundaryValue(response);
            break;
        case "error_guessing":
            // 错误推测测试验证逻辑
            validateErrorHandling(response);
            break;
        default:
            // 默认验证逻辑
            validateBasicResponse(response);
    }
});

function validateEquivalenceClass(response) {
    // 等价类验证：相同等价类的测试结果应该一致
    const expectedCategory = pm.environment.get("expected_category");
    if (expectedCategory === "valid") {
        pm.expect(response.code).to.equal(200);
    } else {
        pm.expect(response.code).to.not.equal(200);
    }
}

function validateBoundaryValue(response) {
    // 边界值验证：关注边界点的特殊处理
    const inputLength = pm.environment.get("input_length");
    const boundary = pm.environment.get("length_boundary");
    
    if (inputLength === boundary) {
        console.log("🎯 边界值测试: 输入长度=" + inputLength);
    }
}

function validateErrorHandling(response) {
    // 错误推测验证：确保错误被正确处理
    const attackVector = pm.environment.get("attack_vector");
    if (attackVector) {
        pm.expect(response.code).to.not.equal(200);
        pm.expect(response.msg).to.not.include(attackVector);
        console.log("🛡️ 安全测试通过: " + attackVector);
    }
}
```
